177 export class CrossCallDedupCache {
178   private readonly windowSize: number;
179   private readonly threshold: number;
183   constructor(opts: CrossCallDedupOptions = {}) {
184     this.windowSize = Math.max(1, opts.windowSize ?? 16);
185     this.threshold = Math.min(1, Math.max(0, opts.threshold ?? 0.85));
186   }
187 
188   /**
189    * Check the cache. Does NOT insert. Returns a replacement string when the
190    * incoming text matches a recent call exactly or fuzzily.
191    */
192   check(text: string): DedupResult {
193     if (this.records.length === 0) return { replacement: null };
194 
195     const exactHash = fnv1a64(text);
196     // Fast path: exact hash match. Walk newest-to-oldest.
197     for (let i = this.records.length - 1; i >= 0; i--) {
198       const r = this.records[i];
199       if (r.exactHash === exactHash) {
200         return {
201           replacement: `[squeez-style: identical to call #${r.callId}${r.label ? ` (${r.label})` : ""}]`,
202           match: { callId: r.callId, kind: "exact", similarity: 1, label: r.label },
203         };
204       }
205     }
206 
207     // Fuzzy path: compute signature, scan window, pick best above threshold.
208     const sig = shingleMinHash(text);
209     let best: { record: CallRecord; sim: number } | null = null;
210     for (let i = this.records.length - 1; i >= 0; i--) {
211       const r = this.records[i];
212       const sim = jaccard(sig, r.signature);
213       if (sim >= this.threshold && (best === null || sim > best.sim)) {
214         best = { record: r, sim };
215       }
216     }
217 
218     if (best !== null) {
219       const pct = Math.round(best.sim * 100);
220       return {
221         replacement: `[squeez-style: ~${pct}% similar to call #${best.record.callId}${best.record.label ? ` (${best.record.label})` : ""}]`,
222         match: {
223           callId: best.record.callId,
224           kind: "fuzzy",
225           similarity: best.sim,
226           label: best.record.label,
227         },
228       };
229     }
230 
231     return { replacement: null };
232   }
233 
234   /**
235    * Insert a new call into the window. Caller decides whether to insert the
239   insert(text: string, label?: string): CallRecord {
240     const record: CallRecord = {
241       callId: this.nextId++,
242       exactHash: fnv1a64(text),
243       signature: shingleMinHash(text),
244       label,
245     };
246     this.records.push(record);
247     if (this.records.length > this.windowSize) {
248       this.records.shift();
249     }
250     return record;
251   }
252 
253   /**
254    * Convenience: check first, then insert (always). Returns the dedup result.
255    * Inserting after a match lets future calls match against the same anchor.
256    */
257   checkAndInsert(text: string, label?: string): DedupResult {
258     const result = this.check(text);
259     this.insert(text, label);
260     return result;
261   }
262 
263   /** Current number of records in the window. */
264   size(): number {
265     return this.records.length;
266   }
267 
268   /** Reset the window. Useful between sessions or for tests. */
269   clear(): void {
270     this.records = [];
271     this.nextId = 1;
272   }
273 }
