16 export function middleware(request: NextRequest) {
17   const { pathname } = request.nextUrl;
18 
19   // Ignora assets estáticos e API routes públicas
20   if (
21     pathname.startsWith('/_next') ||
22     pathname.startsWith('/favicon') ||
23     pathname.includes('.') ||
24     publicApiRoutes.some((r) => pathname.startsWith(r))
25   ) {
26     return NextResponse.next();
27   }
28 
29   // Checa se tem token (via cookie de session ou header)
30   // Em produção, validar JWT server-side aqui
31   const hasSession = request.cookies.has('refresh-token');
32 
33   // Rota pública: permite acesso
34   if (publicRoutes.some((r) => pathname.startsWith(r))) {
35     // Se já autenticado, redireciona pra dashboard
36     if (hasSession && authRoutes.some((r) => pathname.startsWith(r))) {
37       return NextResponse.redirect(new URL('/dashboard', request.url));
38     }
39     return NextResponse.next();
40   }
41 
42   // Rota protegida sem sessão: redireciona pra login
43   if (!hasSession) {
44     const loginUrl = new URL('/login', request.url);
45     loginUrl.searchParams.set('redirect', pathname);
46     return NextResponse.redirect(loginUrl);
47   }
48 
49   // Security headers em todas as respostas
50   const response = NextResponse.next();
51   response.headers.set('X-Frame-Options', 'DENY');
52   response.headers.set('X-Content-Type-Options', 'nosniff');
53   response.headers.set('Referrer-Policy', 'strict-origin-when-cross-origin');
54 
55   return response;
56 }
